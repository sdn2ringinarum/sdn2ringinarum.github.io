8. Integrasi Cloud Gratis Selamanya

Konfigurasi database real-time Firebase Firestore & penyimpanan media Google Drive via Google Apps Script.

Panduan Penyimpanan Foto Gratis di Google Drive (Code.gs)

  1. Buka script.google.com lalu buat Project Baru.
  2. Hapus kode bawaan dan tempel kode doPost(e) di bawah ini (mendukung PNG transparan, JPG, JPEG, WEBP, GIF & otomatis memisahkan folder ke Foto_Website_SDN2Ringinarum / umum / (logo, guru, alumni, galeri)).
  3. Klik Terapkan (Deploy) → Deployment Baru → Jenis: Aplikasi Web.
  4. Atur Yang memiliki akses menjadi Siapa saja (Anyone), lalu salin URL Web App ke kolom di bawah.
const TARGET_FOLDER_ID = "";

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    status: "ok",
    message: "Endpoint Upload Gambar (PNG, JPG, JPEG, WEBP, GIF) SD N 2 Ringinarum aktif."
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var rawBase64 = payload.base64 || "";
    var mimeType = (payload.mimeType || "").toLowerCase().trim();
    var originalName = payload.fileName || payload.filename || ("gambar_" + Date.now());
    var parentSubName = (payload.parentFolder || "umum").trim();
    var categoryName = (payload.category || "umum").toLowerCase().trim();

    if (!rawBase64) throw new Error("Data gambar (base64) kosong.");

    if (rawBase64.indexOf(",") !== -1) {
      var parts = rawBase64.split(",");
      var header = parts[0];
      rawBase64 = parts[1];
      var mimeMatch = header.match(/data:(image\/[a-zA-Z0-9.+-]+);base64/i);
      if (mimeMatch && mimeMatch[1]) mimeType = mimeMatch[1].toLowerCase();
    }

    var extMatch = originalName.match(/\.([a-zA-Z0-9]+)$/);
    var fileExt = extMatch ? extMatch[1].toLowerCase() : "";
    if (!mimeType || mimeType === "application/octet-stream") {
      if (fileExt === "png") mimeType = "image/png";
      else if (fileExt === "webp") mimeType = "image/webp";
      else if (fileExt === "gif") mimeType = "image/gif";
      else mimeType = "image/jpeg";
    }
    if (mimeType === "image/jpg") mimeType = "image/jpeg";

    var allowedMimes = { "image/png": "png", "image/jpeg": "jpg", "image/webp": "webp", "image/gif": "gif" };
    if (!allowedMimes[mimeType]) throw new Error("Format tidak didukung: " + mimeType);

    var cleanBaseName = originalName.replace(/\.[^/.]+$/, "").replace(/[^a-zA-Z0-9_-]/g, "_");
    var finalFileName = cleanBaseName + "_" + Date.now() + "." + allowedMimes[mimeType];

    var rootFolder = TARGET_FOLDER_ID ? DriveApp.getFolderById(TARGET_FOLDER_ID) : getOrCreateSubFolder(DriveApp.getRootFolder(), "Foto_Website_SDN2Ringinarum");
    var umumFolder = getOrCreateSubFolder(rootFolder, parentSubName);
    var targetCategoryFolder = getOrCreateSubFolder(umumFolder, categoryName);

    var decodedBytes = Utilities.base64Decode(rawBase64);
    var blob = Utilities.newBlob(decodedBytes, mimeType, finalFileName);
    var file = targetCategoryFolder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var fileId = file.getId();
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      fileId: fileId,
      folderPath: "Foto_Website_SDN2Ringinarum/" + parentSubName + "/" + categoryName,
      url: "https://lh3.googleusercontent.com/d/" + fileId
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSubFolder(parentFolder, folderName) {
  var folders = parentFolder.getFoldersByName(folderName);
  return folders.hasNext() ? folders.next() : parentFolder.createFolder(folderName);
}
